cmake_minimum_required(VERSION 3.18)
project(cuda_gpu_sorting LANGUAGES CXX)
option(ENABLE_CUDA "Build CUDA vector-add, Bitonic, and radix implementations" ON)
option(CUDA_LINEINFO "Emit CUDA source line information without disabling Release optimization" OFF)
if(ENABLE_CUDA AND CUDA_LINEINFO)
    add_compile_options($<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
endif()

# Set CMAKE_CUDA_COMPILER, CMAKE_CUDA_HOST_COMPILER, and
# CMAKE_CUDA_ARCHITECTURES at configure time when needed.
# Otherwise CMake uses the detected toolkit/compiler defaults.
if(ENABLE_CUDA)
enable_language(CUDA)
add_executable(vector_add vector_add.cu)
set_target_properties(vector_add PROPERTIES
    CXX_STANDARD 14
    CXX_STANDARD_REQUIRED YES
    CUDA_STANDARD 14
    CUDA_STANDARD_REQUIRED YES
)
endif()

if(ENABLE_CUDA)
    add_library(bitonic_sort src/bitonic_sort.cu src/bitonic_shared.cu)
    target_include_directories(bitonic_sort PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
    set_target_properties(bitonic_sort PROPERTIES CUDA_STANDARD 14 CUDA_STANDARD_REQUIRED YES)
    add_library(radix_sort src/radix_sort.cu)
    target_include_directories(radix_sort PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
    set_target_properties(radix_sort PROPERTIES CUDA_STANDARD 14 CUDA_STANDARD_REQUIRED YES)
    option(ENABLE_CUB_BASELINE "Use toolkit CUB headers in the unified runner" ON)
    add_executable(sort_benchmark src/sort_benchmark.cu)
    target_link_libraries(sort_benchmark PRIVATE bitonic_sort radix_sort sorting_support)
    target_compile_definitions(sort_benchmark PRIVATE ENABLE_CUB_BASELINE=$<BOOL:${ENABLE_CUB_BASELINE}>)
    target_compile_definitions(sort_benchmark PRIVATE SORT_BUILD_CONFIGURATION="$<CONFIG>")
    set_target_properties(sort_benchmark PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
endif()

add_library(sorting_support src/sorting_support.cpp src/benchmark_results.cpp)
target_include_directories(sorting_support PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(sorting_support PUBLIC cxx_std_14)
set_target_properties(sorting_support PROPERTIES CXX_EXTENSIONS OFF)

include(CTest)
if(BUILD_TESTING)
    add_executable(cpu_tests tests/cpu_tests.cpp)
    target_link_libraries(cpu_tests PRIVATE sorting_support)
    add_test(NAME cpu_correctness COMMAND cpu_tests)
    add_executable(bitonic_logic_tests tests/bitonic_logic_tests.cpp)
    target_link_libraries(bitonic_logic_tests PRIVATE sorting_support)
    add_test(NAME bitonic_cpu_logic COMMAND bitonic_logic_tests)
    add_executable(bitonic_shared_logic_tests tests/bitonic_shared_logic_tests.cpp)
    target_link_libraries(bitonic_shared_logic_tests PRIVATE sorting_support)
    add_test(NAME bitonic_shared_cpu_logic COMMAND bitonic_shared_logic_tests)
    add_executable(radix_logic_tests tests/radix_logic_tests.cpp)
    target_link_libraries(radix_logic_tests PRIVATE sorting_support)
    add_test(NAME radix_cpu_logic COMMAND radix_logic_tests)
    add_executable(benchmark_options_tests tests/benchmark_options_tests.cpp)
    target_link_libraries(benchmark_options_tests PRIVATE sorting_support)
    add_test(NAME benchmark_options COMMAND benchmark_options_tests)
    if(ENABLE_CUDA)
        if(ENABLE_CUB_BASELINE)
            add_test(NAME all_seven_gpu_smoke COMMAND sort_benchmark --algorithm all --size 1024 --distribution random --seed 12345 --warmups 1 --trials 3)
            add_test(NAME radix_cub_nonpower_gpu_smoke COMMAND sort_benchmark --algorithm radix-1,radix-2,radix-4,radix-8,cub --size 257 --distribution duplicate-heavy --seed 12345 --warmups 1 --trials 3)
        endif()
        add_executable(radix_gpu_tests tests/radix_gpu_tests.cu)
        target_link_libraries(radix_gpu_tests PRIVATE radix_sort sorting_support)
        set_target_properties(radix_gpu_tests PROPERTIES CUDA_STANDARD 14 CUDA_STANDARD_REQUIRED YES)
        add_test(NAME radix_gpu_correctness COMMAND radix_gpu_tests)
        add_executable(bitonic_gpu_tests tests/bitonic_gpu_tests.cu)
        target_link_libraries(bitonic_gpu_tests PRIVATE bitonic_sort sorting_support)
        set_target_properties(bitonic_gpu_tests PROPERTIES CUDA_STANDARD 14 CUDA_STANDARD_REQUIRED YES)
        add_test(NAME bitonic_gpu_correctness COMMAND bitonic_gpu_tests)
        add_executable(bitonic_shared_gpu_tests tests/bitonic_shared_gpu_tests.cu)
        target_link_libraries(bitonic_shared_gpu_tests PRIVATE bitonic_sort sorting_support)
        set_target_properties(bitonic_shared_gpu_tests PROPERTIES CUDA_STANDARD 14 CUDA_STANDARD_REQUIRED YES)
        add_test(NAME bitonic_shared_gpu_correctness COMMAND bitonic_shared_gpu_tests)
        # Deliberate manual comparison, not part of correctness CTest runs.
        add_executable(bitonic_compare tests/bitonic_compare.cu)
        target_link_libraries(bitonic_compare PRIVATE bitonic_sort sorting_support)
        set_target_properties(bitonic_compare PROPERTIES CUDA_STANDARD 14 CUDA_STANDARD_REQUIRED YES)
    endif()
endif()
